\begin{definition}
    the polar form of $f: D \subseteq \mathbb{C} \to \mathbb{C}$ is given 
    by $f(re^{i\theta})=U(r,\theta)+iV(r,\theta)$, where
    $z=re^{i\theta}$, $r\geq0$, and $\theta\in[0,2\pi)$.
    When $f(z)\ne0$, one may also write
    $f(re^{i\theta})=R(r,\theta)e^{i\Phi(r,\theta)}$, where
    $R=|f(z)|$ and $\Phi$ is an argument of $f(z)$.
\end{definition}


\begin{example}
    
    Consider the function $f(z) = z^2$. Then $f(z) = (re^{i\theta})^2 = r^2 e^{i2\theta}$. Thus, the polar form of $f(z)$ is given by $r^2 e^{i2\theta}$.
\end{example}

\begin{definition}
    $\lim_{z \to z_0}f(z)=w_0$ if for every $\epsilon > 0$, there exists a $\delta > 0$ such that 
    $z\in D$ and $0<|z - z_0| < \delta$ imply $|f(z) - w_0| < \epsilon$.
\end{definition}

\begin{remark}
    $\lim_{(x,y) \to (x_0,y_0)}u(x,y)=u_0$ and $\lim_{(x,y) \to (x_0,y_0)}v(x,y)=v_0$ if and only if $\lim_{z \to z_0}f(z)=w_0$.
\end{remark}


\begin{definition}
    
    $f$ is continuous at $z_0$ if $\lim_{z \to z_0}f(z)=f(z_0)$.
\end{definition}

\begin{definition}
    
    $f$ is differentiable at $z_0$ if the limit 
    \[
        f'(z_0) = \lim_{z \to z_0} \frac{f(z) - f(z_0)}{z - z_0}
    \]
    exists.
\end{definition}


\begin{example}
    consder the function $f(z) = |z|^2$ use the definition to find the $f'(0)$.
\end{example}


\begin{solution}

    We have $f(z) = |z|^2 = z\bar{z}$. Then 
    \[
        f'(0)=\lim_{z\to0}\frac{|z|^2}{z}
        =\lim_{z\to0}\bar z=0,
        \qquad |\bar z|=|z|\longrightarrow0.
    \]
    Thus, $f'(0) = 0$.
\end{solution}

\begin{theorem}
    Let $f(z)=u(x,y)+iv(x,y)$ be defined in a neighborhood of
    $z_0=x_0+iy_0$. Suppose the first partial derivatives of $u$ and $v$
    exist and are continuous in that neighborhood. Then $f$ is
    differentiable at $z_0$ if and only if the Cauchy--Riemann equations
    hold at $(x_0,y_0)$:
    \[
        \frac{\partial u}{\partial x} = \frac{\partial v}{\partial y}, \quad \frac{\partial u}{\partial y} = -\frac{\partial v}{\partial x}.
    \]
    
\end{theorem}



\begin{example}
    
    Consider the function $f(z) = z^2$. Then $f(z) = (x + iy)^2 = x^2 - y^2 + 2ixy$. Thus, $u(x,y) = x^2 - y^2$ and $v(x,y) = 2xy$. We have 
    \[
        \frac{\partial u}{\partial x} = 2x, \quad \frac{\partial v}{\partial y} = 2x, \quad \frac{\partial u}{\partial y} = -2y, \quad \frac{\partial v}{\partial x} = 2y.
    \]
    Thus, the Cauchy-Riemann equations are satisfied and $f$ is differentiable at every point in $\mathbb{C}$.
\end{example}

\end{example}

\begin{definition}
    A function $f$ is said to be an analytic at a point $z_0$ if it is differentiable in a neighborhood of $z_0$. 
    If $f$ is analytic at every point in a domain $D$, then $f$ is said to be analytic in $D$.
\end{definition}



\begin{example}
    Consider the function $f(z) = |z|^2$ differentiable at $z_0 = 0$ but not analytic at $z_0 = 0$ since it is not differentiable in any neighborhood of $z_0 = 0$.
\end{example}

\begin{definition}
    Consider The point $z_0$ is said to be a singular point of $f$ if $f$ is not analytic at $z_0$.and there exists a sequence of points $z_n \to z_0$ such that $f$ is analytic at each $z_n$.
\end{definition}

\begin{definition}
    $z_0$ is said to be an isolated singular point of $f$ if $f$ is not analytic at $z_0$ and 
    there exists a neighborhood of $z_0$ in which $f$ is analytic at every point except $z_0$.
\end{definition}


\begin{remark}
    the neighborhood of $z_0$ in which $f$ is analytic at every point except $z_0$ is called a punctured neighborhood of $z_0$.
\end{remark}

    \begin{enumerate}
        \item Consider the function $f(z) = \frac{1}{z}$. Then $f$ is not analytic at $z_0 = 0$ and there exists a punctured neighborhood of $z_0 = 0$ in which $f$ is analytic at every point except $z_0 = 0$. Thus, $z_0 = 0$ is an isolated singular point of $f$.
        \item Consider the function $f(z) = \frac{1}{z^2 + 1}$. Then $f$ is not analytic at $z_0 = i$ and $z_0 = -i$ and there exists a punctured neighborhood of $z_0 = i$ and $z_0 = -i$ in which $f$ is analytic at every point except $z_0 = i$ and $z_0 = -i$.

        \item Consider the function $f(z) = \frac{1}{z^2 + 1}$. Then $f$ is not analytic at $z_0 = i$ and $z_0 = -i$ and there exists a punctured neighborhood of $z_0 = i$ and $z_0 = -i$ in which $f$ is analytic at every point except $z_0 = i$ and $z_0 = -i$. \begin{example}
$f(z) = \frac{1}{\sin(\pi/z)}$. Determine whether $z=0$ is a singular point (isolated or not).
\end{example}
\begin{solution}
    The function is not defined at $z=0$. To find its other singularities, solve
    \[
        \sin\left(\frac{\pi}{z}\right)=0
        \quad\Longleftrightarrow\quad
        \frac{\pi}{z}=n\pi,\qquad n\in\mathbb{Z}.
    \]
    Since $z\ne 0$, the solutions are $z=1/n$ for
    $n\in\mathbb{Z}\setminus\{0\}$. These are poles of $f$, and they
    accumulate at $0$ because $1/n\to 0$ as $|n|\to\infty$.
    Consequently, every punctured neighborhood of $0$ contains singularities
    of $f$. Therefore, $z=0$ is a non-isolated singular point, not an isolated
    singular point.
\end{solution}

\begin{example}
    $f(z)=\log z$. Determine whether $z=0$ is an isolated singularity.
\end{example}
\begin{solution}
    The complex logarithm is multivalued: if $z=re^{i\theta}$, then
    \[
        \log z=\ln r+i(\theta+2\pi k),\qquad k\in\mathbb{Z}.
    \]
    When $z$ travels once around $0$, its argument increases by $2\pi$,
    so the value of the logarithm changes by $2\pi i$. Thus, no
    single-valued analytic branch of $\log z$ exists on a full punctured
    neighborhood of $0$. The point $z=0$ is a branch point (often called a
    logarithmic branch point), not an isolated singularity in the sense
    requiring analyticity throughout a punctured neighborhood. A branch
    such as the principal logarithm is analytic only after a branch cut is
    removed.
\end{solution}

\begin{example}
    Let $f(z)=\operatorname{Log}z$ be the principal logarithm. Determine
    whether $z=0$ is an isolated singularity.
\end{example}
\begin{solution}
    The principal logarithm is defined by
    \[
        \operatorname{Log}z=\ln|z|+i\operatorname{Arg}z,
        \qquad -\pi<\operatorname{Arg}z<\pi,
    \]
    on the slit plane $\mathbb{C}\setminus(-\infty,0]$. It is analytic
    throughout this domain. However, every disk centered at $0$ meets the
    branch cut, so there is no full punctured neighborhood of $0$ on which
    this branch is defined and analytic. Therefore, $z=0$ is not an isolated
    singularity of the principal logarithm in the standard sense; it is a
    branch point. As $z\to0$ within the slit plane,
    $\operatorname{Re}(\operatorname{Log}z)=\ln|z|\to-\infty$.
\end{solution}

\section{Integration}
\label{sec:integration}

Complex integration begins with ordinary real integrals of the real and
imaginary parts. To integrate along a curve in the complex plane, we then
parametrize the curve and include its complex tangent vector. Throughout
this section, curves are oriented: arrows indicate the direction in which
they are traversed.

\subsection{Complex-valued functions of a real variable}

\begin{definition}
    Let $g:[a,b]\to\mathbb C$ have the form $g(t)=u(t)+iv(t)$, where
    $u$ and $v$ are Riemann integrable real-valued functions. Define
    \[
        \boxed{\displaystyle
        \int_a^b g(t)\,dt
        =\int_a^b u(t)\,dt+i\int_a^b v(t)\,dt.}
    \]
    In particular, this integral exists whenever $g$ is continuous on $[a,b]$.
\end{definition}

Linearity and the fundamental theorem of calculus apply componentwise.
For example, if $G:[a,b]\to\mathbb C$ is continuously differentiable, then
\[
    \int_a^b G'(t)\,dt=G(b)-G(a).
\]
We also have the useful inequality
\[
    \left|\int_a^b g(t)\,dt\right|
    \leq\int_a^b |g(t)|\,dt.
\]

\begin{example}
    Evaluate $\displaystyle\int_0^1(t+it^2)\,dt$.
\end{example}
\begin{solution}
    Integrating the two components separately gives
    \[
        \int_0^1(t+it^2)\,dt
        =\left[\frac{t^2}{2}\right]_0^1
         +i\left[\frac{t^3}{3}\right]_0^1
        =\frac12+\frac{i}{3}.
    \]
    The values $g(t)=t+it^2$ trace a parabolic arc in the complex plane.
    The integral is a complex number, represented by the vector to
    $\frac12+\frac{i}{3}$; it is not the length of that arc.
\end{solution}

\subsection{Contours and their parametrizations}

\begin{definition}
    Let $\gamma(t)=x(t)+iy(t)$, $a\leq t\leq b$, where $a<b$.
    The following terminology describes the path and its parametrization.
    \begin{enumerate}
        \item An \emph{arc} (or curve) is a continuous map
        $\gamma:[a,b]\to\mathbb C$; equivalently, both $x$ and $y$
        are continuous on $[a,b]$.
        \item An arc is \emph{simple} if it is injective:
        $\gamma(t_1)\ne\gamma(t_2)$ whenever $t_1\ne t_2$.
        Thus it does not cross itself or retrace a point.
        \item An arc is \emph{closed} if $\gamma(a)=\gamma(b)$.
        It is \emph{simple closed} if this endpoint identification is
        the only repetition: $\gamma$ is injective on $[a,b)$.
        \item A \emph{smooth regular arc} has a continuous derivative
        $\gamma'(t)=x'(t)+iy'(t)$, with one-sided derivatives at the
        endpoints, and $\gamma'(t)\ne0$ for $a<t<b$.
        The nonzero derivative supplies a tangent direction.
        \item A \emph{contour} is a continuous path consisting of finitely
        many smooth regular arcs joined end to end; it is also called
        a piecewise smooth path. Corners at the joins are allowed.
    \end{enumerate}
    The initial and terminal points are $\gamma(a)$ and $\gamma(b)$.
    The orientation is the direction of increasing $t$.
\end{definition}

\begin{remark}
    Simplicity and smoothness describe different properties: simplicity
    concerns repeated points, while smoothness concerns derivatives.
    A simple closed curve is not injective on the full interval $[a,b]$,
    because its endpoints must coincide.
\end{remark}

Writing $\gamma(t)=x(t)+iy(t)$, we have
\[
    \gamma'(t)=x'(t)+iy'(t),\qquad
    L(\gamma)=\int_a^b|\gamma'(t)|\,dt.
\]
For a piecewise smooth contour, the integral for the length is the sum over
its smooth pieces. Two standard parametrizations will be used repeatedly:
\begin{itemize}
    \item The line segment from $z_0$ to $z_1$:
    \[
        \gamma(t)=z_0+t(z_1-z_0),\qquad 0\leq t\leq1.
    \]
    \item The circle with center $c$ and radius $R>0$, traversed once
    counterclockwise:
    \[
        \gamma(t)=c+Re^{it},\qquad 0\leq t\leq2\pi.
    \]
    Restricting the angle interval gives a circular arc. Replacing $e^{it}$
    by $e^{-it}$ gives clockwise traversal of the full circle.
\end{itemize}

\subsection{Definition and calculation of a contour integral}

\begin{definition}
    Let $f$ be continuous on the image of a contour $\gamma:[a,b]\to\mathbb C$.
    The \emph{contour integral} of $f$ along $\gamma$ is
    \[
        \boxed{\displaystyle
        \int_\gamma f(z)\,dz
        =\int_a^b f(\gamma(t))\gamma'(t)\,dt.}
    \]
    If $\gamma$ has several smooth pieces, compute the integral on each
    piece and add the results. For a closed contour, we may write
    $\displaystyle\oint_\gamma f(z)\,dz$.
\end{definition}

To calculate an integral, choose a parametrization with the required
orientation, substitute $z=\gamma(t)$ and $dz=\gamma'(t)\,dt$, and evaluate
the resulting integral over the real parameter interval.

\begin{remark}
    The factor $\gamma'(t)$ is essential:
    $\int_a^b f(\gamma(t))\,dt$ is generally not a contour integral.
    Also, $dz=\gamma'(t)\,dt$ differs from the arc-length element
    $ds=|\gamma'(t)|\,dt$.
\end{remark}

For $f=u+iv$ and $z=x+iy$, multiplication gives the equivalent formula
\[
    \int_\gamma f(z)\,dz
    =\int_\gamma (u\,dx-v\,dy)
     +i\int_\gamma (v\,dx+u\,dy).
\]
Here $u$ and $v$ are evaluated at the points $(x(t),y(t))$ of the contour.

\begin{example}
    Evaluate $\displaystyle\int_\gamma z\,dz$, where $\gamma$ is the
    straight segment from $0$ to $1+i$.
\end{example}
\begin{solution}
    Set $\gamma(t)=(1+i)t$, $0\leq t\leq1$. Then
    $\gamma'(t)=1+i$, so
    \[
        \int_\gamma z\,dz
        =\int_0^1 (1+i)t(1+i)\,dt
        =\int_0^1 2it\,dt=i.
    \]
\end{solution}

\subsection{Orientation, addition, and a circular contour}

Contour integration is linear in the integrand. It is unchanged by a
piecewise smooth, orientation-preserving reparametrization of the same
traversal. If $-\gamma$ denotes the reverse traversal and
$\gamma_1*\gamma_2$ means first following $\gamma_1$ and then $\gamma_2$
(with matching endpoints), then
\[
    \int_{-\gamma} f(z)\,dz=-\int_\gamma f(z)\,dz,
    \qquad
    \int_{\gamma_1*\gamma_2}f(z)\,dz
    =\int_{\gamma_1}f(z)\,dz+\int_{\gamma_2}f(z)\,dz.
\]

\begin{example}
    Evaluate $\displaystyle\oint_{C_R}\frac{1}{z}\,dz$, where
    $C_R$ is the circle $|z|=R$, $R>0$, traversed once counterclockwise.
\end{example}
\begin{solution}
    Parametrize by $z=Re^{it}$, $0\leq t\leq2\pi$, so that
    $dz=iRe^{it}\,dt$. Therefore
    \[
        \oint_{C_R}\frac{1}{z}\,dz
        =\int_0^{2\pi}\frac{iRe^{it}}{Re^{it}}\,dt
        =\int_0^{2\pi}i\,dt=2\pi i.
    \]
    Clockwise traversal gives $-2\pi i$. Although $1/z$ is undefined at
    the center, it is continuous at every point of the contour, so the
    integral is well-defined. Its value does not depend on $R$.
\end{solution}

\subsection{Path dependence and antiderivatives}

\begin{example}
    Compare $\displaystyle\int_\gamma\bar z\,dz$ along two paths from
    $0$ to $1+i$: the diagonal segment $\gamma_d$, and the broken path
    $\gamma_b$ going from $0$ to $1$ and then from $1$ to $1+i$.
\end{example}
\begin{solution}
    On the diagonal, $z=(1+i)t$, $0\leq t\leq1$, and hence
    \[
        \int_{\gamma_d}\bar z\,dz
        =\int_0^1 (1-i)t(1+i)\,dt
        =\int_0^1 2t\,dt=1.
    \]
    On the horizontal part of the broken path, use $z=t$, $0\leq t\leq1$:
    \[
        \int_0^1 t\,dt=\frac12.
    \]
    On the vertical part, use $z=1+it$, $0\leq t\leq1$, so $dz=i\,dt$:
    \[
        \int_0^1 (1-it)i\,dt
        =\int_0^1(t+i)\,dt=\frac12+i.
    \]
    Thus $\int_{\gamma_b}\bar z\,dz=1+i$, whereas
    $\int_{\gamma_d}\bar z\,dz=1$. The endpoints alone do not determine
    a contour integral.
\end{solution}

\begin{definition}
    An \emph{antiderivative} (or \emph{primitive}) of $f$ on an open set
    $D\subseteq\mathbb C$ is an analytic function $F:D\to\mathbb C$
    such that $F'(z)=f(z)$ for every $z\in D$.
\end{definition}

\begin{theorem}[Fundamental theorem for contour integrals]
    Suppose $f$ is continuous on an open set $D$ and has a primitive $F$
    there. For every contour $\gamma:[a,b]\to D$,
    \[
        \boxed{\displaystyle
        \int_\gamma f(z)\,dz=F(\gamma(b))-F(\gamma(a)).}
    \]
    Consequently, the integral depends only on the endpoints, and the
    integral around every closed contour in $D$ is zero.
\end{theorem}
\begin{proof}
    On each smooth piece the chain rule gives
    $(F\circ\gamma)'(t)=F'(\gamma(t))\gamma'(t)
    =f(\gamma(t))\gamma'(t)$. Integrating and adding the endpoint
    differences over the pieces proves the formula.
\end{proof}

\begin{example}
    Evaluate $\displaystyle\int_\gamma z\,dz$ along any contour from
    $0$ to $1+i$, and compare with the straight-segment calculation.
\end{example}
\begin{solution}
    The function $F(z)=z^2/2$ is a primitive of $z$ on all of $\mathbb C$.
    Therefore, regardless of the chosen contour,
    \[
        \int_\gamma z\,dz=\frac{(1+i)^2}{2}-0=i.
    \]
    This agrees with the earlier parametrization. In particular, the
    diagonal and broken paths in Figure~\ref{fig:integration-path-dependence}
    both give $i$ for this integrand, even though they give different values
    for the integrand $\bar z$.
\end{solution}

\begin{remark}
    Analyticity on a domain alone does not guarantee a primitive on that
    entire domain. The function $1/z$ is analytic on
    $\mathbb C\setminus\{0\}$, but its integral around $C_R$ is $2\pi i$,
    so it has no primitive there. On the slit plane
    $\mathbb C\setminus(-\infty,0]$, however, $\operatorname{Log}z$ is a
    primitive of $1/z$. A full circle around $0$ is not contained in this
    slit plane, so the endpoint formula using $\operatorname{Log}$ cannot
    be applied to that circle.
\end{remark}

\subsection{Estimating integrals: the ML inequality}

\begin{theorem}[ML inequality]
    Let $f$ be continuous on a contour $\gamma$ of length $L$. If
    $|f(z)|\leq M$ at every point of the contour, then
    \[
        \boxed{\displaystyle\left|\int_\gamma f(z)\,dz\right|\leq ML.}
    \]
\end{theorem}
\begin{proof}
    Using the parametrized integral and the triangle inequality gives
    \[
        \left|\int_\gamma f(z)\,dz\right|
        \leq\int_a^b |f(\gamma(t))|\,|\gamma'(t)|\,dt
        \leq M\int_a^b|\gamma'(t)|\,dt=ML.
    \]
\end{proof}

\begin{example}
    Find an upper bound for
    $\displaystyle\left|\int_\gamma\frac{1}{z-2}\,dz\right|$,
    where $\gamma(t)=e^{it}$, $0\leq t\leq\pi$, is the upper unit
    semicircle from $1$ to $-1$.
\end{example}
\begin{solution}
    On $\gamma$, $|z|=1$, so the reverse triangle inequality gives
    \[
        |z-2|\geq 2-|z|=1,
        \qquad \left|\frac{1}{z-2}\right|\leq1.
    \]
    We may take $M=1$. Since $|\gamma'(t)|=|ie^{it}|=1$, the length is
    $L=\int_0^\pi1\,dt=\pi$. Hence
    \[
        \left|\int_\gamma\frac{1}{z-2}\,dz\right|\leq\pi.
    \]
    This is an estimate, not a claim that the integral has modulus $\pi$.
\end{solution}

\noindent
When working with a new contour integral, first check the path and its
orientation, and check that the integrand is defined on the path. Then
choose between direct parametrization, a primitive valid on a domain
containing the whole path, or an ML estimate if only a bound is needed.
